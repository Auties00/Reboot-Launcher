cmake_minimum_required(VERSION 3.30)
project(server_browser VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

option(SB_BUILD_NET "Build the QUIC edge, client and apps (needs MsQuic)" ON)
option(SB_WITH_NATS "Build the NATS backbone (needs cnats)" ON)
option(SB_BUILD_TESTS "Build unit and integration tests" ON)
option(SB_BUILD_FUZZ "Build libFuzzer targets (Clang only)" OFF)
option(SB_USE_MIMALLOC "Link mimalloc as the global allocator" ON)
option(SB_FRAME_POINTERS "Keep frame pointers so perf can unwind production stacks (~1% cost)" ON)
set(SB_SANITIZE "" CACHE STRING "Sanitizers to enable, e.g. address,undefined or thread")

include(SbFeatures)
include(SbWarnings)

find_package(Threads REQUIRED)
find_package(OpenSSL 3.0 REQUIRED)
find_package(Boost REQUIRED CONFIG COMPONENTS headers pfr container unordered)
find_package(roaring CONFIG REQUIRED)
find_package(zstd CONFIG REQUIRED)
find_package(tomlplusplus CONFIG REQUIRED)
find_package(CLI11 CONFIG REQUIRED)
if(SB_USE_MIMALLOC)
  find_package(mimalloc CONFIG REQUIRED)
endif()
if(SB_BUILD_NET)
  find_package(MsQuic REQUIRED)
endif()
if(SB_WITH_NATS)
  find_package(cnats CONFIG REQUIRED)
endif()

if(SB_FRAME_POINTERS)
  add_compile_options(-fno-omit-frame-pointer -mno-omit-leaf-frame-pointer)
endif()
if(SB_SANITIZE)
  add_compile_options(-fsanitize=${SB_SANITIZE} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${SB_SANITIZE})
endif()

add_subdirectory(src)
if(SB_BUILD_NET)
  add_subdirectory(apps)
endif()
if(SB_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(SB_BUILD_FUZZ)
  add_subdirectory(fuzz)
endif()
